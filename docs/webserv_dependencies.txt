  THE NATURAL DEPENDENCY ORDER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  nothing works without config
  config is useless without a loop to run it
  the loop is useless without HTTP to process
  HTTP is useless without responses to build and send

  so the only logical sequential order is:

  DEV A → config + sockets + loop engine
  DEV B → HTTP parsing + routing + response building
  DEV C → filesystem + CGI + final integration

  DEV B cannot start until A's shared types exist as stubs
  DEV C cannot start until B's Router and HttpResponse exist

  BEFORE ANYONE STARTS — day 1 together (1 hour)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  write and commit empty stub .hpp files
  no implementation — just field names and method signatures
  everyone compiles against the same headers from day 1

  stubs needed:
  ├── ListenEntry.hpp
  ├── LocationConfig.hpp
  ├── ServerConfig.hpp
  ├── AHandler.hpp
  ├── HttpRequest.hpp     empty feed() and isComplete()
  ├── HttpResponse.hpp    empty serialize()
  └── Router.hpp          route() returns hardcoded 200 OK


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  DEV A — CONFIG + SOCKETS + REACTOR LOOP
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  owns and builds:
  ├── ConfigParser
  ├── ListenEntry
  ├── LocationConfig
  ├── ServerConfig
  ├── AHandler
  ├── Reactor
  ├── AcceptHandler
  └── ClientHandler (shell — recv/send/close, calls B's stubs)

  delivers:
      a running server that accepts connections and returns
      a hardcoded "HTTP/1.0 200 OK\r\n\r\nhi" to any request

  depends on from others:
      HttpRequest::feed()       stub — just returns false
      HttpRequest::isComplete() stub — just returns false
      Router::route()           stub — just returns 200 OK
      HttpResponse::serialize() stub — just returns the hardcoded string

  does NOT need:
      any real HTTP parsing
      any routing logic
      any filesystem access


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  DEV B — HTTP PARSING + ROUTING + RESPONSE BUILDING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  owns and builds:
  ├── HttpRequest        full state machine implementation
  ├── HttpResponse       full serialize() implementation
  └── Router             full route() implementation
                         calls C's FileSystem stubs for file/upload/delete

  delivers:
      route() that correctly parses requests, matches config,
      returns accurate HttpResponse for GET/POST/DELETE/errors

  depends on from others:
      ServerConfig / LocationConfig    from A (read-only)
      FileSystem::readFile()           stub — returns dummy string
      FileSystem::writeFile()          stub — returns true
      FileSystem::deleteFile()         stub — returns true
      FileSystem::autoindex()          stub — returns dummy HTML

  does NOT need:
      poll(), sockets, or any fd logic
      filesystem implementation
      CGI implementation


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  DEV C — FILESYSTEM + CGI + INTEGRATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  owns and builds:
  ├── FileSystem         readFile, writeFile, deleteFile,
  │                      autoindex, mimeType
  └── CgiHandler         AHandler subclass, fork/pipe/execve/timeout

  delivers:
      real files served over GET
      uploads written to disk
      files deleted
      CGI scripts executed and output returned

  depends on from others:
      AHandler            from A — CgiHandler inherits it
      Reactor.add()       from A — to register pipe fd
      ClientHandler*      from A — writes into _writeBuf
      HttpRequest.body    from B — POST body piped to CGI stdin
      Router signals      from B — "CGI needed" path triggers fork()

  does NOT need:
      HTTP parsing
      routing logic
      poll() loop internals

  SHARED DATA STRUCTURES — what they REALLY need to contain
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  these are the types everyone touches
  fields must be agreed and frozen before day 1 coding starts
  no one is allowed to add fields after that without team agreement

  ┌──────────────────────────────────────────────────────────────┐
  │  ListenEntry                                                 │
  │  ──────────────────────────────────────────────────────────  │
  │  string   host     "127.0.0.1" or "0.0.0.0"                 │
  │  int      port     8080                                      │
  │                                                              │
  │  used by:  A (socket setup), A (AcceptHandler knows ip:port) │
  │            B (Router filters configs by ip:port)             │
  └──────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────┐
  │  LocationConfig                                              │
  │  ──────────────────────────────────────────────────────────  │
  │  string              path                                    │
  │  bool                isExact                                 │
  │  string              root                                    │
  │  vector      index                                   │
  │  vector      allowedMethods                          │
  │  bool                autoindex                               │
  │  map  cgiExtensions    ext → interpreter path │
  │  string              uploadStore                             │
  │  int                 redirectCode     0 = not a redirect     │
  │  string              redirectUri                             │
  │                                                              │
  │  used by:  A (ConfigParser fills it)                         │
  │            B (Router reads it to make decisions)             │
  │            C (FileSystem uses root, uploadStore, cgiExt)     │
  └──────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────┐
  │  ServerConfig                                                │
  │  ──────────────────────────────────────────────────────────  │
  │  vector     listens                             │
  │  vector          serverNames                         │
  │  size_t                  maxBodySize    bytes, not string     │
  │  map         errorPages     {404: "/path"}       │
  │  vector  locations                           │
  │                                                              │
  │  used by:  A (ConfigParser fills it, AcceptHandler holds ref)│
  │            B (Router reads it entirely)                      │
  │            C (FileSystem reads errorPages for error bodies)  │
  └──────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────┐
  │  HttpRequest                                                 │
  │  ──────────────────────────────────────────────────────────  │
  │  string              method                                  │
  │  string              uri                                     │
  │  string              version                                 │
  │  map  headers                                 │
  │  string              body                                    │
  │  ParseState          state                                   │
  │  size_t              contentLength                           │
  │  ──────────────────────────────────────────────────────────  │
  │  bool  feed(string chunk)    advance parser, return complete │
  │  bool  isComplete()          true when state == COMPLETE     │
  │                                                              │
  │  used by:  A (ClientHandler holds it, calls feed/isComplete) │
  │            B (owns the implementation of feed/isComplete)    │
  │            C (reads body for CGI stdin, reads headers)       │
  └──────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────┐
  │  HttpResponse                                                │
  │  ──────────────────────────────────────────────────────────  │
  │  int                 statusCode                              │
  │  string              statusMsg                               │
  │  map  headers                                 │
  │  string              body                                    │
  │  ──────────────────────────────────────────────────────────  │
  │  string  serialize()    produces raw HTTP/1.0 bytes          │
  │                         always includes Connection: close    │
  │                         always includes Content-Length       │
  │                                                              │
  │  used by:  A (ClientHandler stores it, calls serialize())    │
  │            B (Router builds and returns it)                  │
  │            C (CgiHandler fills body, B adds headers)         │
  └──────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────┐
  │  AHandler                                                    │
  │  ──────────────────────────────────────────────────────────  │
  │  int  _fd                                                    │
  │  ──────────────────────────────────────────────────────────  │
  │  virtual void  handle_input()   = 0                          │
  │  virtual void  handle_output()  = 0                          │
  │  virtual void  handle_close()   = 0                          │
  │  virtual       ~AHandler()                                   │
  │                                                              │
  │  used by:  A (defines it, Reactor holds map) │
  │            C (CgiHandler inherits it)                        │
  └──────────────────────────────────────────────────────────────┘

  DAY 1 STUBS (all together)
       │
       ▼
  DEV A builds config + loop
       │  delivers: loop running, hardcoded response
       │  delivers: ServerConfig / LocationConfig frozen
       ▼
  DEV B builds HTTP + Router
       │  depends on: ServerConfig / LocationConfig shapes (A)
       │  delivers: real responses, Router::route() works
       │  delivers: FileSystem interface agreed with C
       ▼
  DEV C builds FileSystem + CGI
            depends on: Router::route() CGI signal (B)
            depends on: AHandler (A)
            depends on: ClientHandler._writeBuf accessible (A)
            delivers: real files, uploads, CGI output
