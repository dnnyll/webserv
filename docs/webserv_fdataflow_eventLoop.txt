│     │

│     │
│     └── revents & POLLIN on clientFd
│           └── ClientHandler::handleRead()
│                 ├── recv(clientFd, buf, 4096)
│                 │     └── raw bytes land in buf[]
│                 │
│                 └── _request.getData(buf)
│                       ├── _buffer += chunk
│                       │
│                       └── while state != COMPLETE:
│                             decode()
│                               │
│                               ├── case REQUEST_LINE
│                               │     └── decodeRequestLine()
│                               │           ├── skip leading \r\n
│                               │           ├── extractRequestLine()
│                               │           │     └── find \r\n
│                               │           │         extract line
│                               │           │         erase from _buffer
│                               │           ├── splitRequestLine()
│                               │           │     ├── find 1st space → method
│                               │           │     ├── find 2nd space → uri
│                               │           │     └── remainder      → version
│                               │           ├── validateRequestLine()
│                               │           │     ├── method ∈ {GET,POST,DELETE}
│                               │           │     ├── uri[0] == '/'
│                               │           │     └── version == "HTTP/1.1"
│                               │           └── _state = HEADERS
│                               │
│                               ├── case HEADERS
│                               │     └── decodeHeaders()
│                               │           └── loop:
│                               │                 find \r\n
│                               │                 if pos == 0 → blank line
│                               │                   └── resolveBodyState()
│                               │                         ├── chunked?  → _state = CHUNKED
│                               │                         ├── length?   → _state = BODY
│                               │                         └── neither   → _state = COMPLETE
│                               │                 else → split ": "
│                               │                        headers[key] = value
│                               │                        erase from _buffer
│                               │
│                               ├── case BODY
│                               │     └── decodeBody()
│                               │           ├── bytesNeeded = contentLength - _bodyBytesRead
│                               │           ├── toRead = min(needed, available)
│                               │           ├── body += _buffer.substr(0, toRead)
│                               │           ├── _buffer.erase(0, toRead)
│                               │           └── if _bodyBytesRead == contentLength
│                               │                 └── _state = COMPLETE
│                               │
│                               └── case CHUNKED
│                                     └── decodeChunked()
│                                           └── loop:
│                                                 parseChunkSize()
│                                                   ├── find \r\n
│                                                   ├── extract hex size → _chunkSize
│                                                   └── if _chunkSize == 0
│                                                         └── _state = COMPLETE
│                                                 wait until buffer >= _chunkSize + 2
│                                                 body += _buffer.substr(0, _chunkSize)
│                                                 _buffer.erase(0, _chunkSize + 2)
│                                                 _chunkSize = 0 → repeat
│
│     └── revents & POLLOUT on clientFd
│           └── ClientHandler::handleWrite()

│                 ├── _outBuffer.erase(0, bytesSent)
│                 └── if _outBuffer.empty()
│                       ├── _keepAlive → _request.reset() → wait for next request
│                       └── !_keepAlive → _isClosed = true
